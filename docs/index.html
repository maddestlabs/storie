# Bouncing Character Demo This is a simple demo of the S|torie engine using SES sandboxing. ## Game Code ```javascript // Game state let x = 40; let y = 12; let vx = 0.5; let vy = 0.3; let frame = 0; function init() { console.log('🎮 Game initialized!'); term.layerID = 'default'; } function update(delta) { // Update position x += vx * delta * 60; y += vy * delta * 60; // Bounce off walls if (x <= 0 || x >= termCanvas.width() - 1) { vx = -vx; x = Math.max(0, Math.min(termCanvas.width() - 1, x)); } if (y <= 0 || y >= termCanvas.height() - 1) { vy = -vy; y = Math.max(0, Math.min(termCanvas.height() - 1, y)); } // Input if (key.pressed(key.SPACE)) { vx = (Math.random() - 0.5) * 2; vy = (Math.random() - 0.5) * 2; } frame++; } function render() { // Clear screen term.clear(); // Draw border for (let i = 0; i < termCanvas.width(); i++) { termCanvas.plot(i, 0, '═', {r: 100, g: 100, b: 100}); termCanvas.plot(i, termCanvas.height() - 1, '═', {r: 100, g: 100, b: 100}); } for (let i = 0; i < termCanvas.height(); i++) { termCanvas.plot(0, i, '║', {r: 100, g: 100, b: 100}); termCanvas.plot(termCanvas.width() - 1, i, '║', {r: 100, g: 100, b: 100}); } // Corners termCanvas.plot(0, 0, '╔', {r: 100, g: 100, b: 100}); termCanvas.plot(termCanvas.width() - 1, 0, '╗', {r: 100, g: 100, b: 100}); termCanvas.plot(0, termCanvas.height() - 1, '╚', {r: 100, g: 100, b: 100}); termCanvas.plot(termCanvas.width() - 1, termCanvas.height() - 1, '╝', {r: 100, g: 100, b: 100}); // Draw character const ix = Math.floor(x); const iy = Math.floor(y); termCanvas.plot(ix, iy, '🚀', {r: 255, g: 255, b: 0}); // Draw trail const trailLen = 5; for (let i = 1; i <= trailLen; i++) { const tx = Math.floor(x - vx * i * 2); const ty = Math.floor(y - vy * i * 2); const alpha = 1 - (i / trailLen); const c = Math.floor(255 * alpha); if (tx >= 0 && tx < termCanvas.width() && ty >= 0 && ty < termCanvas.height()) { termCanvas.plot(tx, ty, '·', {r: c, g: c, b: 0}); } } // Draw info term.write(2, 2, `Frame: ${frame}`, {r: 150, g: 150, b: 150}); term.write(2, 3, `FPS: ${Math.round(1 / getDelta())}`, {r: 150, g: 150, b: 150}); term.write(2, 4, `Pos: (${ix}, ${iy})`, {r: 150, g: 150, b: 150}); term.write(2, termCanvas.height() - 2, `Press SPACE to randomize velocity`, {r: 100, g: 200, b: 255}); } ``` ## About This demo showcases: - **SES Sandboxing**: JavaScript code runs in an isolated compartment - **Main Loop**: Separate init/update/render handlers - **Canvas API**: Drawing primitives and text - **Input**: Keyboard interaction
